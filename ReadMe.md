function validateSNiP(object, ruleId) {
  const rule = snipRules[ruleId];
  if (!rule) return { status: 'unknown' };
  
  const errors = [];
  for (const [key, check] of Object.entries(rule.checks)) {
    if (object.parameters[key] !== undefined) {
      const result = check(object.parameters[key]);
      if (result !== 'OK') errors.push(result);
    }
  }
  
  return {
    status: errors.length === 0 ? 'passed' : 'failed',
    errors: errors
  };
}

// 1. Выполнение в правильном порядке (топологическая сортировка)
function executeGraph(nodes, connections) {
  const order = topologicalSort(nodes, connections);
  const results = {};
  
  for (const nodeId of order) {
    const node = nodes[nodeId];
    const inputs = node.inputs.map(inp => results[inp.sourceId]);
    
    // Вызов ноды
    results[nodeId] = node.execute(inputs);
  }
  
  return results[order[order.length - 1]]; // финальный результат
}

javascript
// Проверка коллизий 10 000 объектов на GPU (0.1 сек)
// Вместо 30 секунд на CPU

class WebGPUCollisionDetector {
  async init() {
    const adapter = await navigator.gpu.requestAdapter();
    this.device = await adapter.requestDevice();
    this.createPipeline();
  }
  
  checkCollisions(objects) {
    // Загрузка объектов в GPU-память (один раз)
    const buffer = this.device.createBuffer({
      size: objects.length * 32, // AABB
      usage: GPUBufferUsage.STORAGE
    });
    
    // Выполнение Compute Shader
    const pass = this.device.createCommandEncoder();
    const computePass = pass.beginComputePass();
    computePass.setPipeline(this.pipeline);
    computePass.setBindGroup(0, bindGroup);
    computePass.dispatchWorkgroups(Math.ceil(objects.length / 256));
    computePass.end();
    
    // Чтение результатов обратно на CPU
